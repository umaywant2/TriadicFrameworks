function interpret(raw_query):

    envelope = {}

    envelope.intent        = ask("What is the intended outcome?")
    envelope.regime        = ask("What regime applies?")
    envelope.scale         = ask("What scale is relevant?")
    envelope.transition    = ask("What transition type is present?")
    envelope.boundary      = ask("What boundary condition applies?")
    envelope.invariants    = ask_list("What invariants must hold?")
    envelope.modifiers     = ask_list("What modifiers influence the system?")
    envelope.substrate     = ask("What substrate is involved?")
    envelope.lineage       = ask_list("What upstream dependencies exist?")
    envelope.failure_mode  = ask("What is the failure mode?")
    envelope.time_regime   = ask("What is the time-regime?")
    envelope.symmetry      = ask("What symmetry or asymmetry exists?")
    envelope.raw_query     = raw_query

    return envelope
