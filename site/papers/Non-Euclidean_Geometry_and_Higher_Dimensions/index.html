Progress Function¶

function getBadgeProgress(validatorStats) {
  const progress = {};
  for (const badge in badgeCriteria) {
    const criteria = badgeCriteria[badge];
    const keys = Object.keys(criteria);
    const met = keys.every(k => validatorStats[k] >= criteria[k]);
    const percent = Math.min(100, Math.round((validatorStats[keys[0]] / criteria[keys[0]]) * 100));
    progress[badge] = { earned: met, percent };
  }
  return progress;
}
